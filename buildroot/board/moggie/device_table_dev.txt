# Moggie static /dev -- used because the 2.6.35 kernel has no CONFIG_DEVTMPFS.
#
# <name>	<type>	<mode>	<uid>	<gid>	<major>	<minor>	<start>	<inc>	<count>
#
# Modes are deliberately stricter than a desktop /dev: /dev/null etc. are
# 0600 here because these nodes live on a read-only squashfs and every
# process on the speaker runs as root. At boot rcS mounts a tmpfs over /dev
# and mdev recreates the normal modes there.

/dev/null	c	600	0	0	1	3	-	-	-
